  const bsScript = document.createElement('script');

  const App = {
    masterBBM: [], allHistoryData: [], charts: {}, cacheData: {},
    
    // --- FITUR TEMA GELAP/TERANG ---
    initTheme: function() {
      const savedTheme = localStorage.getItem('theme') || 'light';
      document.documentElement.setAttribute('data-bs-theme', savedTheme);
      const icon = document.getElementById('theme-icon');
      if(savedTheme === 'dark' && icon) icon.classList.replace('bi-moon-stars-fill', 'bi-sun-fill');
      this.updateChartColors(savedTheme);
    },

    toggleTheme: function() {
      const html = document.documentElement;
      const currentTheme = html.getAttribute('data-bs-theme');
      const newTheme = currentTheme === 'dark' ? 'light' : 'dark';
      
      html.setAttribute('data-bs-theme', newTheme);
      localStorage.setItem('theme', newTheme);
      
      const icon = document.getElementById('theme-icon');
      if(newTheme === 'dark') icon.classList.replace('bi-moon-stars-fill', 'bi-sun-fill');
      else icon.classList.replace('bi-sun-fill', 'bi-moon-stars-fill');
      
      this.updateChartColors(newTheme);
    },

    updateChartColors: function(theme) {
      Chart.defaults.color = theme === 'dark' ? '#adb5bd' : '#666';
      Chart.defaults.borderColor = theme === 'dark' ? '#333' : '#e5e5e5';
      Object.values(this.charts).forEach(chart => chart.update());
    },

    init: function() {
      this.initTheme(); // Panggil fungsi tema saat aplikasi dimuat
      
      let now = new Date();
      now.setMinutes(now.getMinutes() - now.getTimezoneOffset());
      document.getElementById('tanggal').value = now.toISOString().slice(0, 16);

      google.script.run
        .withSuccessHandler(this.onDataLoaded.bind(this))
        .withFailureHandler(this.onSystemError.bind(this))
        .getDropdownData();
    },

    switchView: function(viewId, navElement) {
      if (viewId === 'settings') {
        Swal.fire({
          title: '<i class="bi bi-shield-lock-fill text-danger" style="font-size:2rem;"></i><br>Akses Admin',
          text: 'Masukkan PIN rahasia untuk membuka menu ini.',
          input: 'password',
          inputPlaceholder: 'Masukkan PIN...',
          inputAttributes: { autocapitalize: 'off', autocorrect: 'off' },
          showCancelButton: true,
          confirmButtonColor: '#0d6efd',
          confirmButtonText: 'Buka Kunci',
          cancelButtonText: 'Batal'
        }).then((result) => {
          if (result.isConfirmed) {
            if (result.value === '609215') {
              this.executeSwitch(viewId, navElement);
            } else {
              Swal.fire({ icon: 'error', title: 'Akses Ditolak', text: 'PIN yang Anda masukkan salah!' });
            }
          }
        });
      } else {
        this.executeSwitch(viewId, navElement);
      }
    },

    executeSwitch: function(viewId, navElement) {
      document.querySelectorAll('.view-section').forEach(el => el.classList.remove('active'));
      document.getElementById('view-' + viewId).classList.add('active');
      document.querySelectorAll('.nav-item').forEach(el => el.classList.remove('active'));
      navElement.classList.add('active');

      if(viewId === 'history' || viewId === 'settings') this.loadHistory();
      if(viewId === 'dashboard') this.loadDashboard();
    },

    onDataLoaded: function(response) {
      if (response.status === 'error') return this.onSystemError(response.message);
      const data = response.data;
      this.cacheData = data;
      
      this.populate('plat', data.plat, 'Pilih Plat');
      this.populate('supir', data.supir, 'Pilih Supir');
      this.populate('spbu', data.spbu, 'Pilih Lokasi');
      
      this.masterBBM = data.bbm; 
      const tipeUnik = [...new Set(data.bbm.map(item => item[1]).filter(String))];
      this.populate('tipeBBM', tipeUnik, 'Pilih Tipe BBM');
      
      this.renderSettingsLists(); 
      document.getElementById('globalLoader').style.display = 'none';
    },

    populate: function(id, arrayData, placeholder) {
      const el = document.getElementById(id);
      el.innerHTML = `<option value="" disabled selected>${placeholder}</option>`;
      arrayData.forEach(item => el.innerHTML += `<option value="${item}">${item}</option>`);
    },

    renderSettingsLists: function() {
      const buildTable = (headers, dataRows, sheetName, type) => {
        let ths = headers.map(h => `<th>${h}</th>`).join('');
        let html = `<div class="table-responsive"><table class="table table-sm table-bordered text-center align-middle mb-0" style="font-size:0.75rem; white-space:nowrap;">
          <thead class="table-light"><tr>${ths}<th>Aksi</th></tr></thead><tbody>`;

        if(!dataRows || dataRows.length === 0) {
           html += `<tr><td colspan="${headers.length + 1}" class="text-secondary py-3">Belum ada data / Kosong</td></tr>`;
        } else {
           dataRows.forEach(item => {
             let tds = headers.map((_, i) => {
               if(type === 'BBM' && i === 2) return `<td class="text-danger fw-bold">Rp ${Number(item.data[i]||0).toLocaleString('id-ID')}</td>`;
               return `<td>${item.data[i] || '-'}</td>`;
             }).join('');
             
             html += `<tr>${tds}
               <td>
                 <button class="btn btn-sm btn-outline-primary py-0 px-1 me-1" onclick="App.editMaster('${sheetName}', ${item.rowId}, '${type}')"><i class="bi bi-pencil"></i></button>
                 <button class="btn btn-sm btn-outline-danger py-0 px-1" onclick="App.deleteMaster('${sheetName}', ${item.rowId})"><i class="bi bi-trash"></i></button>
               </td></tr>`;
           });
        }
        return html + `</tbody></table></div>`;
      };

      document.getElementById('listSupir').innerHTML = buildTable(['ID Supir', 'Nama Supir', 'No Telp', 'Status'], this.cacheData.supirFull, 'DATA SUPIR', 'SUPIR');

      document.getElementById('listBbm').innerHTML = buildTable(['Jenis Bahan Bakar', 'Tipe BBM', 'Harga per Liter'], this.cacheData.bbmFull, 'MASTER BBM', 'BBM');

               else Swal.fire('Gagal', res.message, 'error');
             }).deleteMasterDataAction(sheetName, rowId);
          }
       });
    },

    addMasterData: function(kategori, inputId) {
      const inputEl = document.getElementById(inputId);
      const nilai = inputEl.value.trim();
      if(!nilai) return Swal.fire('Perhatian', 'Input kosong!', 'warning');
      
      Swal.fire({ title: 'Menyimpan...', didOpen: () => Swal.showLoading() });
      google.script.run.withSuccessHandler(res => {
        if(res.status === 'success') {
          Swal.fire('Berhasil!', res.message, 'success');
          inputEl.value = '';
          this.init(); 
        }
      }).addMasterData(kategori, nilai);
    },

    onTipeChange: function() {
      const tipe = document.getElementById('tipeBBM').value;
      const jenisEl = document.getElementById('jenisBBM');
      jenisEl.disabled = false;
      jenisEl.innerHTML = '<option value="" disabled selected>Pilih Jenis BBM</option>';
      this.masterBBM.filter(item => item[1] === tipe).forEach(item => {
        jenisEl.innerHTML += `<option value="${item[0]}">${item[0]}</option>`;
      });
      document.getElementById('harga').value = ''; 
    },

    onJenisChange: function() {
      const jenis = document.getElementById('jenisBBM').value;
      const bbmData = this.masterBBM.find(item => item[0] === jenis);
      if(bbmData && bbmData[2]) document.getElementById('harga').value = bbmData[2];
      else document.getElementById('harga').value = 0;
    },

    loadHistory: function() {
      const tbody = document.getElementById('history-table-body');
      tbody.innerHTML = `<tr><td colspan="7" class="text-center py-4"><div class="spinner-border spinner-border-sm me-2"></div>Memuat...</td></tr>`;
      
      google.script.run
        .withSuccessHandler(res => {
          if(res.status === 'error') {
            return tbody.innerHTML = `<tr><td colspan="7" class="text-danger text-center"><i class="bi bi-exclamation-triangle-fill me-2"></i>Error: ${res.message}</td></tr>`;
          }
          this.allHistoryData = res.data;
          
          const bulans = [...new Set(res.data.map(d => d.bulan))];
          let bulanHtml = '<option value="ALL">Semua Bulan</option>';
          bulans.forEach(b => bulanHtml += `<option value="${b}">${b}</option>`);
          document.getElementById('filterBulan').innerHTML = bulanHtml;
          
          this.applyHistoryFilter();
        })
        .withFailureHandler(err => {
          tbody.innerHTML = `<tr><td colspan="7" class="text-danger text-center"><i class="bi bi-x-circle-fill me-2"></i>Gagal Memuat: ${err.message}</td></tr>`;
        })
        .getHistoryData();
    },

    applyHistoryFilter: function() {
      const bulanVal = document.getElementById('filterBulan').value;
      const tbody = document.getElementById('history-table-body');
      
      let filtered = this.allHistoryData;
      if (bulanVal !== 'ALL') filtered = filtered.filter(d => d.bulan === bulanVal);

      if(filtered.length === 0) return tbody.innerHTML = `<tr><td colspan="6" class="text-center text-secondary py-3">Tidak ada data.</td></tr>`;
      
      tbody.innerHTML = filtered.map(trx => `
        <tr>
          <td>${trx.tanggal}</td>
          <td class="fw-bold text-dark">${trx.plat}</td>
          <td>${trx.supir}</td>
          <td><span class="badge bg-secondary">${trx.volume} L</span></td>
          <td class="fw-bold text-danger">Rp ${trx.total.toLocaleString('id-ID')}</td>
          <td>${trx.odo.toLocaleString('id-ID')}</td>
        </tr>`).join('');
    },

    loadDashboard: function() {

        const d = res.data;
        const fmtRp = val => 'Rp ' + Math.round(val || 0).toLocaleString('id-ID');
        
        document.getElementById('dash-vol').innerText = (d.stats.globalVol || 0).toLocaleString('id-ID', {minimumFractionDigits: 1, maximumFractionDigits: 2});
        document.getElementById('dash-biaya').innerText = fmtRp(d.stats.globalBiaya);
        document.getElementById('dash-jarak').innerText = d.stats.globalJarak ? d.stats.globalJarak.toLocaleString('id-ID') : '0'; 
        document.getElementById('dash-kml').innerText = (d.stats.globalKML || 0).toFixed(2);
        document.getElementById('dash-costkm').innerText = fmtRp(d.stats.globalCostKM);
        
        const setTrend = (id, pct, currVal, fmtFn) => {
          const el = document.getElementById(id);
          let trend = '';
          if(pct > 0) trend = `<span class="trend-up"><i class="bi bi-arrow-up-right"></i> Naik ${pct.toFixed(1)}%</span>`;
          else if(pct < 0) trend = `<span class="trend-down"><i class="bi bi-arrow-down-right"></i> Turun ${Math.abs(pct).toFixed(1)}%</span>`;
          else trend = `<span class="trend-flat"><i class="bi bi-dash"></i> Stabil 0%</span>`;
          
          el.innerHTML = `<span class="text-muted" style="font-size:0.7rem;">Bulan Ini: <b class="text-dark">${fmtFn(currVal)}</b><br>${trend} vs bln lalu</span>`;
        };
        
        setTrend('trend-vol', d.stats.volPct || 0, d.stats.currVol || 0, v => v.toFixed(2) + ' L');
        setTrend('trend-biaya', d.stats.biayaPct || 0, d.stats.currBiaya || 0, v => fmtRp(v));

        let persenKML = d.stats.persenKML || "0.0";
        let standarAvg = d.stats.standarAvg || "0.00";

             <span class="${kmlColor} fw-bold me-2" style="font-size:0.95rem;">${kmlIcon} ${persenKML}%</span>
           </div>
           <span style="font-size:0.8rem; color:#6c757d; display:block; margin-top:2px;">
             vs Standar (${standarAvg}) &bull; <strong class="text-dark">${kmlText}</strong>
           </span>`;

        let isJarakLebih = d.stats.isJarakLebih;
        let persenJarak = d.stats.persenJarak || "0.0";
        let standarJarak = Math.round(d.stats.standarJarak || 0).toLocaleString('id-ID');
        let selisihJarak = Math.round(d.stats.selisihJarak || 0).toLocaleString('id-ID');
        let jarakColor = isJarakLebih ? 'text-success' : 'text-danger';
        let jarakIcon = isJarakLebih ? '↑' : '↓';
        let jarakText = isJarakLebih ? `Lebih jauh +${selisihJarak} KM` : `Lebih pendek -${selisihJarak} KM`;

        document.getElementById('trend-jarak').innerHTML =
          `<div class="d-flex align-items-center mt-1">
             <span class="${jarakColor} fw-bold me-2" style="font-size:0.95rem;">${jarakIcon} ${persenJarak}%</span>
           </div>
           <span style="font-size:0.8rem; color:#6c757d; display:block; margin-top:2px;">
             vs Target (${standarJarak} KM) &bull; <strong class="text-dark">${jarakText}</strong>
           </span>`;

        let isCostLebihHemat = d.stats.isCostLebihHemat;
        let persenCost = d.stats.persenCostKM || "0.0";
        let standarCost = d.stats.standarCostKM || 0;
        let selisihCost = d.stats.selisihCostKM || 0;
        let costColor = isCostLebihHemat ? 'text-success' : 'text-danger'; 
        let costIcon = isCostLebihHemat ? '↓' : '↑'; 
        let costText = isCostLebihHemat 
          ? `Lebih hemat Rp ${Math.round(selisihCost).toLocaleString('id-ID')}/KM` 
          : `Lebih mahal Rp ${Math.round(selisihCost).toLocaleString('id-ID')}/KM`;

        document.getElementById('trend-costkm').innerHTML = 
          `<div class="d-flex align-items-center mt-1">
             <span class="${costColor} fw-bold me-2" style="font-size:0.95rem;">${costIcon} ${persenCost}%</span>
           </div>
           <span style="font-size:0.8rem; color:#6c757d; display:block; margin-top:2px;">
             vs Standar (${fmtRp(standarCost)}) &bull; <strong class="text-dark">${costText}</strong>
           </span>`;

        let validData = (d.analysisTable || []).filter(x => x.totalJarak > 0 && x.actualKML > 0);
        validData.sort((a,b) => b.actualKML - a.actualKML);

        // Fungsi pembuat tabel Top 5
        const buildTopTable = (data, isIrit) => {
          let colorClass = isIrit ? 'text-success' : 'text-danger';
          let html = `<table class="table table-sm table-borderless mb-0 align-middle">
                        <thead>
                          <tr class="text-secondary border-bottom">
                            <th>No</th>
                            <th>No Polisi</th>
                            <th>Jenis</th>
                            <th class="text-center">Standar KM/L</th>
                            <th class="text-end">Aktual KM/L</th>
                          </tr>
                        </thead>
                        <tbody>`;
          data.forEach((x, i) => {
             html += `<tr class="border-bottom">
                        <td><b>${i+1}</b></td>
                        <td class="fw-bold">${x.plat}</td>
                        <td>${x.jenis || '-'}</td>
                        <td class="text-center">${x.stdMin || 0}</td>
                        <td class="${colorClass} fw-bold text-center">${x.actualKML.toFixed(2)}</td>
                      </tr>`;
          });
          html += `</tbody></table>`;
          return html;
        };

        // Ambil 5 teratas (Irit) dan 5 terbawah (Boros)
        let dataIrit = validData.slice(0, 5);
        let dataBoros = validData.slice(-5).reverse();

        document.getElementById('top-irit').innerHTML = dataIrit.length > 0 ? buildTopTable(dataIrit, true) : '<span class="text-muted">Data tidak cukup</span>';
        document.getElementById('top-boros').innerHTML = dataBoros.length > 0 ? buildTopTable(dataBoros, false) : '<span class="text-muted">Data tidak cukup</span>';

        let supirHtml = '';
        if (d.supirStats) {
          // Mengubah object menjadi array dan memfilter nama supir yang tidak valid/kosong/header
          let driverArray = Object.keys(d.supirStats)
            .filter(name => name && name.trim() !== '' && name.toUpperCase() !== 'NAMA SUPIR')
            .map(name => {
              let s = d.supirStats[name];
              let totalJarak = parseFloat(s.totalJarak) || 0;
              let totalVol = parseFloat(s.totalVol) || 0;
              let totalBiaya = parseFloat(s.biaya) || 0;

              let avgKML = totalVol > 0 ? (totalJarak / totalVol) : 0;
              let avgCostKM = totalJarak > 0 ? (totalBiaya / totalJarak) : 0;

              return { name, totalJarak, avgKML, avgCostKM };
            });

          // Urutkan berdasarkan KM/L tertinggi (paling efisien)
          driverArray.sort((a, b) => b.avgKML - a.avgKML);

          driverArray.forEach((ds, index) => {
            let rankBadge = index + 1;
            if (index === 0) rankBadge = '🥇 1';
            else if (index === 1) rankBadge = '🥈 2';
            else if (index === 2) rankBadge = '🥉 3';

            // Status evaluasi berdasarkan jarak tempuh operasional & efisiensi
            let statusBadge = '<span class="badge bg-secondary">Evaluasi</span>';
            if (ds.totalJarak > 0) {
              if (ds.avgKML < 7) {
                statusBadge = '<span class="badge bg-danger">Boros</span>';
              } else if (ds.avgKML >= 7 && ds.avgKML < 9) {
                statusBadge = '<span class="badge bg-primary">Normal</span>';
              } else {
                statusBadge = '<span class="badge bg-success">Efisien</span>';
              }
            }

            supirHtml += `
              <tr class="text-center">
                <td class="fw-bold">${rankBadge}</td>
                <td class="text-start fw-bold">${ds.name}</td>
                <td>${Math.round(ds.totalJarak).toLocaleString('id-ID')}</td>
                <td class="fw-bold text-info">${ds.avgKML.toFixed(2)}</td>
                <td class="fw-bold text-warning">Rp ${Math.round(ds.avgCostKM).toLocaleString('id-ID')}</td>
                <td>${statusBadge}</td>
              </tr>
            `;
          });
        }
        document.getElementById('supir-body').innerHTML = supirHtml || '<tr><td colspan="6" class="text-center text-muted">Data supir tidak tersedia</td></tr>';

        this.renderAnalysisTable(d.analysisTable || []);
        // this.renderPivotTable(d.pivotData);
        this.renderCharts(d.chartMonthly, d.pivotData, d.bbmDist, d.analysisTable);

      }).getDashboardData();
    },
    
    renderAnalysisTable: function(data) {
      const tbody = document.getElementById('analysis-body');
      if (data.length === 0) return tbody.innerHTML = `<tr><td colspan="12" class="text-center py-4">Data belum cukup untuk dianalisa.</td></tr>`;
      
      let html = '';
      data.forEach(d => {
        let statusHtml = '';
        if (d.stdMin === 0) {
           statusHtml = '<span class="badge bg-secondary">NO DATA</span>';
        } else if (d.actualKML < d.stdMin) {
           statusHtml = '<span class="badge bg-danger">BOROS</span>';
        } else {
           statusHtml = '<span class="badge bg-success">EFISIEN</span>';
        }

        html += `
        <tr>
          <td class="fw-bold text-start">${d.plat}</td>
          <td>${d.jenis}</td>
          <td class="text-secondary fw-bold">${d.stdMin}</td>
          <td class="text-center">${statusHtml}</td>
          <td>${d.kmAwal.toLocaleString('id-ID')}</td>
          <td>${d.kmAkhir.toLocaleString('id-ID')}</td>
          <td class="fw-bold">${d.totalJarak.toLocaleString('id-ID')}</td>
          <td>${d.jarakPerHari.toFixed(1)}</td>
          <td>${d.totalVol.toFixed(2)}</td>
          <td>Rp ${Math.round(d.totalBiaya).toLocaleString('id-ID')}</td>
          <td class="fw-bold text-primary">${d.actualKML.toFixed(2)}</td>
          <td>Rp ${Math.round(d.actualCostKM).toLocaleString('id-ID')}</td>
        </tr>`;
      });
      tbody.innerHTML = html;
    },

    renderPivotTable: function(pData) {
      if (!pData || !pData.months) return;
      const thead = document.getElementById('pivot-head');
      const tbody = document.getElementById('pivot-body');
      
      let headHtml = `<tr><th rowspan="2" class="align-middle text-start bg-light">Plat Mobil</th>`;
      pData.months.forEach(m => headHtml += `<th colspan="2" class="text-center bg-primary text-white border-light">${m}</th>`);
      headHtml += `</tr><tr>`;
      pData.months.forEach(m => headHtml += `<th class="text-center bg-light" style="font-size:0.75rem;">Vol (L)</th><th class="text-center bg-light" style="font-size:0.75rem;">Total (Rp)</th>`);
      headHtml += `</tr>`;
      thead.innerHTML = headHtml;

      let bodyHtml = '';
      Object.keys(pData.cars).forEach(plat => {
        bodyHtml += `<tr><td class="fw-bold text-start">${plat}</td>`;
        pData.months.forEach(m => {
          const d = pData.cars[plat][m];
          let vol = 0;
          if (d && d.costkm > 0 && d.kml > 0) vol = (d.biaya / d.costkm) / d.kml;
          
          let volTxt = vol > 0 ? vol.toFixed(1) : '-';
          let biayaTxt = d && d.biaya > 0 ? Math.round(d.biaya).toLocaleString('id-ID') : '-';
          
          bodyHtml += `<td class="text-center text-secondary">${volTxt}</td><td class="text-end text-danger fw-bold">${biayaTxt}</td>`;
        });
        bodyHtml += `</tr>`;
      });
      tbody.innerHTML = bodyHtml;
    },

    renderCharts: function(monthly, pData, bbmDist, analysisTable) {
      const render = (id, type, labels, datasets, opts) => {
        if(this.charts[id]) this.charts[id].destroy();
        this.charts[id] = new Chart(document.getElementById(id).getContext('2d'), {
          type: type, data: { labels: labels, datasets: datasets }, options: opts
        });
      };

      render('chartMonthly', 'bar', monthly.labels, [
        { 
          label: 'Biaya (Rp)', 
          data: monthly.biaya, 
          backgroundColor: 'rgba(176, 38, 255, 0.7)', // Neon Purple transparan
          borderColor: '#b026ff',                     // Garis pinggir Neon Purple tebal
          borderWidth: 2,
          borderRadius: 6,                            // Ujung batang membulat modern
          yAxisID: 'y' 
        },
        { 
          label: 'Volume (L)', 
          data: monthly.vol, 
          type: 'line', 
          borderColor: '#00f3ff',                     // Neon Cyan Terang
          borderWidth: 3, 
          tension: 0.4,                               // <--- INI YANG MEMBUAT GARISNYA SMOOTH / MELENGKUNG
          pointBackgroundColor: '#121212',            // Titik tengah warna gelap
          pointBorderColor: '#00f3ff',                // Pinggiran titik warna Neon Cyan
          pointBorderWidth: 2,
          pointRadius: 4,                             // Ukuran titik
          pointHoverRadius: 7,                        // Titik membesar saat disentuh
          yAxisID: 'y1' 
        }
      ], {
        responsive: true, interaction: { mode: 'index', intersect: false },
        scales: { 
          y: { type: 'linear', position: 'left' }, 
          y1: { type: 'linear', position: 'right', grid: { drawOnChartArea: false } } 
        }
      });

      let effDatasets = []; let costDatasets = [];
      const colors = ['#0d6efd', '#198754', '#dc3545', '#ffc107', '#0dcaf0', '#6f42c1', '#fd7e14', '#20c997'];
      let cIdx = 0;
      
      Object.keys(pData.cars).forEach(plat => {
         let dataEff = []; let dataCost = [];
         pData.months.forEach(m => {
             let d = pData.cars[plat][m];
             dataEff.push(d && d.kml > 0 ? d.kml.toFixed(2) : null);
             dataCost.push(d && d.costkm > 0 ? d.costkm.toFixed(0) : null);
         });
         let color = colors[cIdx % colors.length];
         effDatasets.push({ label: plat, data: dataEff, borderColor: color, backgroundColor: color, fill: false, tension: 0.3, spanGaps: true });
         costDatasets.push({ label: plat, data: dataCost, borderColor: color, backgroundColor: color, fill: false, tension: 0.3, spanGaps: true });
         cIdx++;
      });

      render('chartEfficiency', 'line', pData.months, effDatasets, 
        { responsive: true, plugins: { legend: { position: 'bottom', labels: { boxWidth: 12 } } }, scales: { y: { beginAtZero: true } } }
      );
      render('chartCostKm', 'line', pData.months, costDatasets, 
        { responsive: true, plugins: { legend: { position: 'bottom', labels: { boxWidth: 12 } } }, scales: { y: { beginAtZero: true } } }
      );

      if (bbmDist) {
        // 1. FILTER DATA: Membuang label "JENIS BAHAN BAKAR" dan baris kosong
        let bbmLabels = [];
        let bbmData = [];
        Object.keys(bbmDist).forEach(k => {
          if (k && k.trim() !== "" && k.toUpperCase() !== "JENIS BAHAN BAKAR") {
            bbmLabels.push(k);
            bbmData.push(bbmDist[k].vol.toFixed(1));
          }
        });

        // 2. WARNA LIGHT NEON
        let pieColors = ['#00f3ff', '#00ff66', '#ff00ff', '#ccff00', '#b026ff', '#ff6600'];
        
        if(this.charts['chartBbmDist']) this.charts['chartBbmDist'].destroy();
        this.charts['chartBbmDist'] = new Chart(document.getElementById('chartBbmDist').getContext('2d'), {
          type: 'doughnut',
          data: { 
            labels: bbmLabels, 
            datasets: [{ 
              data: bbmData, 
              backgroundColor: pieColors,
              borderWidth: 2,
              borderColor: 'transparent', // Garis batas transparan agar warna menyatu
              hoverOffset: 6
            }] 
          },
          options: { 
            responsive: true, 
            maintainAspectRatio: false, 
            cutout: '45%', // Donat lebih tipis dan modern
            plugins: { 
              legend: { 
                position: 'right',
                labels: {
                  usePointStyle: true, // Indikator legend menjadi bulat
                  padding: 15
                }
              },
              tooltip: {
                callbacks: {
                  label: function(context) {
                    let label = context.label || '';
                    let value = parseFloat(context.raw) || 0;
                    // Hitung total semua BBM
                    let total = context.dataset.data.reduce((a, b) => parseFloat(a) + parseFloat(b), 0);
                    // Hitung persentase
                    let percentage = ((value / total) * 100).toFixed(1) + '%';
                    
                    // Format tampilan: "Nama BBM: 100 L (50.5%)"
                    return label + ': ' + value.toLocaleString('id-ID') + ' L (' + percentage + ')';
                  }
                }
              } 
            } 
          }
        });
      }
    },

    onImageSelected: function(event) {
      const file = event.target.files[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = e => {
        document.getElementById('previewImage').src = e.target.result;
        document.getElementById('imageString').value = e.target.result; 
        document.getElementById('cameraBtn').style.display = 'none';
        document.getElementById('previewContainer').style.display = 'block';
      };
      reader.readAsDataURL(file);
    },

    removeImage: function() {
      document.getElementById('fotoStruk').value = '';
      document.getElementById('imageString').value = '';
      document.getElementById('previewContainer').style.display = 'none';
      document.getElementById('cameraBtn').style.display = 'block';
    },

    submitData: function() {
      const form = document.getElementById('formBBM');
      const imgString = document.getElementById('imageString').value;

      if (!form.checkValidity() || !imgString) {
        form.classList.add('was-validated');
        if (!imgString) Swal.fire('Perhatian', 'Wajib melampirkan foto struk!', 'warning');
        return;
      }

      const btn = document.getElementById('btnSubmit');
      btn.disabled = true;
      btn.innerHTML = '<span class="spinner-border spinner-border-sm me-2"></span>Memproses...';

      const payload = {
        tanggal: document.getElementById('tanggal').value,
        plat: document.getElementById('plat').value,
        supir: document.getElementById('supir').value,
        spbu: document.getElementById('spbu').value,
        tipeBBM: document.getElementById('tipeBBM').value,
        jenisBBM: document.getElementById('jenisBBM').value,
        harga: document.getElementById('harga').value,
        volume: document.getElementById('volume').value,
        odometer: document.getElementById('odometer').value,
        imageString: imgString
      };

      google.script.run
        .withSuccessHandler((res) => {
          btn.disabled = false;
          btn.innerHTML = '<i class="bi bi-cloud-arrow-up-fill me-2"></i>Kirim Data Transaksi';
          if (res.status === 'success') {
            Swal.fire({ icon: 'success', title: 'Tersimpan!', timer: 2000, showConfirmButton: false });
            form.reset();
            form.classList.remove('was-validated');
            this.removeImage();
            let now = new Date();
            now.setMinutes(now.getMinutes() - now.getTimezoneOffset());
            document.getElementById('tanggal').value = now.toISOString().slice(0, 16);
            document.getElementById('jenisBBM').disabled = true; 
            this.switchView('history', document.querySelectorAll('.nav-item')[2]);
          } else {
            Swal.fire('Gagal Menyimpan', res.message, 'error');
          }
        })
        .withFailureHandler((err) => {
          btn.disabled = false;
          btn.innerHTML = '<i class="bi bi-cloud-arrow-up-fill me-2"></i>Kirim Data Transaksi';
          this.onSystemError(err.message);
        })
        .simpanData(payload);
    },

    onSystemError: function(msg) {
      document.getElementById('globalLoader').style.display = 'none';
      Swal.fire('Kesalahan Sistem', msg, 'error');
    }
  };

  document.addEventListener('DOMContentLoaded', () => App.init());
